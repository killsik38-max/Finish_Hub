-- Обновление значений FPS
local lastTick = tick()
local frameCount = 0
RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    local currentTick = tick()
    if currentTick - lastTick >= 0.5 then
        local fps = math.floor(frameCount / (currentTick - lastTick) + 0.5)
        local fpsString = "FPS: " .. tostring(fps)
        fCardVal.Text = tostring(fps) .. " FPS"
        ScreenFpsWidget.Text = fpsString
        frameCount = 0
        lastTick = currentTick
    end
end)

local function createToggle(langKey, defaultState, callback)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -6, 0, 46)
    frame.BackgroundColor3 = Color3.fromRGB(24, 24, 32)
    frame.Parent = ScrollingFrame
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = frame
    
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(45, 45, 60)
    stroke.Thickness = 1
    stroke.Parent = frame
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.65, 0, 1, 0)
    label.Position = UDim2.new(0, 14, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = getLoc(langKey)
    label.TextColor3 = Color3.fromRGB(240, 240, 245)
    label.TextSize = 13
    label.Font = Enum.Font.GothamMedium
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = frame
    
    table.insert(UI13Texts, {type = "toggle", key = langKey, label = label})
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 64, 0, 28)
    btn.Position = UDim2.new(1, -74, 0.5, -14)
    btn.BackgroundColor3 = Color3.fromRGB(45, 45, 60)
    btn.Text = ""
    btn.Parent = frame
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(1, 0)
    btnCorner.Parent = btn
    
    local circle = Instance.new("Frame")
    circle.Size = UDim2.new(0, 22, 0, 22)
    circle.Position = UDim2.new(0, 3, 0.5, -11)
    circle.BackgroundColor3 = Color3.fromRGB(220, 220, 230)
    circle.Parent = btn
    
    local circleCorner = Instance.new("UICorner")
    circleCorner.CornerRadius = UDim.new(1, 0)
    circleCorner.Parent = circle
    
    local active = false
    
    local function setToggleState(state)
        if active == state then return end
        active = state
        
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
            circle.Position = UDim2.new(1, -25, 0.5, -11)
            circle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            task.spawn(function()
                callback(function() return active end)
            end)
        else
            btn.BackgroundColor3 = Color3.fromRGB(45, 45, 60)
            circle.Position = UDim2.new(0, 3, 0.5, -11)
            circle.BackgroundColor3 = Color3.fromRGB(220, 220, 230)
        end
    end
    
    btn.MouseButton1Click:Connect(function()
        setToggleState(not active)
    end)
    
    if defaultState then
        setToggleState(true)
    end
    
    return frame, btn, function() return active end
end

local function getUniqueColor(str)
    local hash = 0
    for i = 1, #str do
        hash = (hash * 31 + string.byte(str, i)) % 360
    end
    return Color3.fromHSV(hash / 360, 0.9, 1)
end

RunService.Heartbeat:Connect(function()
    if not eggActive then return end
    local folder = Workspace:FindFirstChild("RenderedEggs", true)
    if not folder then return end
    
    for _, egg in ipairs(folder:GetChildren()) do
        if egg:IsA("Model") or egg:IsA("BasePart") then
            if selectedEggTypes[egg.Name] == nil then
                selectedEggTypes[egg.Name] = true
                
                local checkBtn = Instance.new("TextButton")
                checkBtn.Size = UDim2.new(1, 0, 0, 24)
                checkBtn.BackgroundColor3 = Color3.fromRGB(30, 180, 90)
                checkBtn.Text = "  [✓] " .. egg.Name
                checkBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
                checkBtn.TextSize = 11
                checkBtn.Font = Enum.Font.GothamMedium
                checkBtn.TextXAlignment = Enum.TextXAlignment.Left
                checkBtn.Parent = eggListContainer
                
                local checkCorner = Instance.new("UICorner")
                checkCorner.CornerRadius = UDim.new(0, 4)
                checkCorner.Parent = checkBtn
                
                eggListContainer.CanvasSize = UDim2.new(0, 0, 0, eggListLayout.AbsoluteContentSize.Y)
                
                checkBtn.MouseButton1Click:Connect(function()
                    selectedEggTypes[egg.Name] = not selectedEggTypes[egg.Name]
                    if selectedEggTypes[egg.Name] then
                        checkBtn.BackgroundColor3 = Color3.fromRGB(30, 180, 90)
                        checkBtn.Text = "  [✓] " .. egg.Name
                    else
                        checkBtn.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
                        checkBtn.Text = "  [X] " .. egg.Name
                    end
                end)
            end
            
            if selectedEggTypes[egg.Name] == true then
                if not eggData[egg] then
                    local data = {}
                    local highlight = Instance.new("Highlight")
                    highlight.Adornee = egg
                    highlight.FillColor = getUniqueColor(egg.Name)
                    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                    highlight.Parent = ScreenGui
                    data.Highlight = highlight
                    
                    local billboard = Instance.new("BillboardGui")
                    billboard.Adornee = egg:IsA("Model") and (egg.PrimaryPart or egg:FindFirstChildWhichIsA("BasePart")) or egg
                    billboard.Size = UDim2.new(0, 200, 0, 50)
                    billboard.StudsOffset = Vector3.new(0, 3, 0)
                    billboard.AlwaysOnTop = true
                    billboard.MaxDistance = 100000
                    
                    local textLabel = Instance.new("TextLabel")
                    textLabel.Size = UDim2.new(1, 0, 1, 0)
                    textLabel.BackgroundTransparency = 1
                    textLabel.Text = egg.Name
                    textLabel.TextColor3 = highlight.FillColor
                    textLabel.TextStrokeTransparency = 0
                    textLabel.TextSize = 14
                    textLabel.Font = Enum.Font.GothamBold
                    textLabel.Parent = billboard
                    
                    billboard.Parent = ScreenGui
                    data.Billboard = billboard
                    eggData[egg] = data
                end
            else
                if eggData[egg] then
                    if eggData[egg].Highlight then eggData[egg].Highlight:Destroy() end
                    if eggData[egg].Billboard then eggData[egg].Billboard:Destroy() end
                    eggData[egg] = nil
                end
            end
        end
    end
end)

-- 4. Instant Proximity Prompt
local originalPromptData = {}
createToggle("instPrompt", false, function(isActive)
    while isActive() do
        pcall(function()
            for _, v in pairs(Workspace:GetDescendants()) do
                if not isActive() then break end
                if v:IsA("ProximityPrompt") then
                    if not originalPromptData[v] then
                        originalPromptData[v] = {HoldDuration = v.HoldDuration, MaxActivationDistance = v.MaxActivationDistance}
                    end
                    v.HoldDuration = 0
                end
            end
        end)
        local t = 0
        while t < 1 and isActive() do task.wait(0.2) t = t + 0.2 end
    end
    pcall(function()
        for v, data in pairs(originalPromptData) do
            if v and v.Parent then
                v.HoldDuration = data.HoldDuration
                v.MaxActivationDistance = data.MaxActivationDistance
            end
        end
        originalPromptData = {}
    end)
end)

createToggle("autoRebuild", false, function(isActive)
    pcall(function()
        local remotes = ReplicatedStorage:FindFirstChild("Remotes", true) and ReplicatedStorage.Remotes:FindFirstChild("Game")
        if remotes then
            if remotes:FindFirstChild("RebirthHintSeen") then remotes.RebirthHintSeen:FireServer() end
            if remotes:FindFirstChild("Rebirth") then remotes.Rebirth:FireServer() end
        end
    end)
    while isActive() do
        local elapsed = 0
        while elapsed < 1800 and isActive() do task.wait(1) elapsed = elapsed + 1 end
        if not isActive() then break end
        pcall(function()
            local remotes = ReplicatedStorage:FindFirstChild("Remotes", true) and ReplicatedStorage.Remotes:FindFirstChild("Game")
            if remotes then
                if remotes:FindFirstChild("RebirthHintSeen") then remotes.RebirthHintSeen:FireServer() end
                if remotes:FindFirstChild("Rebirth") then remotes.Rebirth:FireServer() end
            end
        end)
    end
end)

-- 6. Auto Upgrade Base (Normal)
createToggle("autoUpNorm", false, function(isActive)
    while isActive() do
        pcall(function()
            for _, remote in pairs(ReplicatedStorage:GetDescendants()) do
                if not isActive() then break end
                if remote:IsA("RemoteEvent") and (string.lower(remote.Name):find("upgrade") or string.lower(remote.Name):find("buy") or string.lower(remote.Name):find("farm")) then
                    remote:FireServer()
                elseif remote:IsA("RemoteFunction") and (string.lower(remote.Name):find("upgrade") or string.lower(remote.Name):find("buy") or string.lower(remote.Name):find("farm")) then
                    remote:InvokeServer()
                end
            end
            for _, v in pairs(Workspace:GetDescendants()) do
                if not isActive() then break end
                if v:IsA("ProximityPrompt") and (string.lower(v.ActionText):find("upgrade") or string.lower(v.ObjectText):find("upgrade") or string.lower(v.Name):find("upgrade")) then
                    fireproximityprompt(v)
                end
            end
        end)
        if not isActive() then break end
        local t = 0
        while t < 1 and isActive() do task.wait(0.2) t = t + 0.2 end
    end
end)

                if v:IsA("ProximityPrompt") and (string.lower(v.ActionText):find("upgrade") or string.lower(v.ObjectText):find("upgrade") or string.lower(v.Name):find("upgrade") or string.lower(v.ActionText):find("max")) then
                    fireproximityprompt(v)
